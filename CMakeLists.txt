# Icons are embedded so the app shows its icon without being installed. kevcpp-ui-icons.gresource.xml
# (toolbar and panel icons) is generated by data/art/make_icons.py.
find_program(GLIB_COMPILE_RESOURCES glib-compile-resources REQUIRED)
set(KEVC_RESOURCE_C)
foreach(bundle kevcpp kevcpp-ui-icons)
  set(xml ${CMAKE_SOURCE_DIR}/data/${bundle}.gresource.xml)
  set(out ${CMAKE_BINARY_DIR}/${bundle}-resources.c)
  execute_process(
    COMMAND ${GLIB_COMPILE_RESOURCES} --generate-dependencies ${xml}
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}/data
    OUTPUT_VARIABLE deps_raw)
  string(REPLACE "\n" ";" deps "${deps_raw}")
  list(FILTER deps EXCLUDE REGEX "^$")
  list(TRANSFORM deps PREPEND ${CMAKE_SOURCE_DIR}/data/)
  string(MAKE_C_IDENTIFIER ${bundle} c_name)
  add_custom_command(
    OUTPUT ${out}
    COMMAND ${GLIB_COMPILE_RESOURCES} --generate-source --target=${out} --c-name=${c_name}
            --sourcedir=${CMAKE_SOURCE_DIR}/data ${xml}
    DEPENDS ${xml} ${deps}
    COMMENT "Compiling GResources (${bundle})")
  list(APPEND KEVC_RESOURCE_C ${out})
endforeach()

# kevc_version.h: the commit this build is made from (its date, time and hash are the version).
# Checked on every build; the header only changes, and BuildInfo.cpp only recompiles, on a new commit.
add_custom_target(kevc_version
  COMMAND ${CMAKE_COMMAND} -DSOURCE_DIR=${CMAKE_SOURCE_DIR} -DOUTPUT=${CMAKE_BINARY_DIR}/generated/kevc_version.h
          -P ${CMAKE_SOURCE_DIR}/cmake/version.cmake
  BYPRODUCTS ${CMAKE_BINARY_DIR}/generated/kevc_version.h
  COMMENT "Reading the version from git")
add_dependencies(kev-cpp kevc_version)
target_include_directories(kev-cpp PRIVATE ${CMAKE_BINARY_DIR}/generated)

# ELF binaries can't embed an icon, but file managers that honour GIO's metadata::custom-icon (Nemo)
# show one. Set it after every build; harmless where gio or a desktop session isn't available.
find_program(GIO_EXECUTABLE gio)
if(GIO_EXECUTABLE)
  add_custom_command(TARGET kev-cpp POST_BUILD
    COMMAND sh -c "\"$0\" set \"$1\" metadata::custom-icon \"$2\" 2>/dev/null || true"
            ${GIO_EXECUTABLE} $<TARGET_FILE:kev-cpp>
            "file://${CMAKE_SOURCE_DIR}/data/icons/hicolor/256x256/apps/org.kevcpp.KevCpp.png"
    VERBATIM)
endif()

include(GNUInstallDirs)
install(TARGETS kev-cpp RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
# `kev PATH` opens a project or folder from a terminal; it finds kev-cpp beside itself.
install(PROGRAMS data/kev DESTINATION ${CMAKE_INSTALL_BINDIR})
# Only the app icon goes into the system icon theme; the kevc-* action icons stay embedded.
install(DIRECTORY data/icons/ DESTINATION ${CMAKE_INSTALL_DATADIR}/icons
        PATTERN "actions" EXCLUDE)
# The launcher points at the installed binary by full path (~/.local/bin is often not on the desktop
# session's PATH). Written at install time so `cmake --install --prefix ...` gets the right path.
install(CODE "
  set(prefix \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}\")
  file(READ \"${CMAKE_SOURCE_DIR}/data/org.kevcpp.KevCpp.desktop.in\" desktop)
  string(REPLACE \"@BINDIR@\" \"\${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_BINDIR}\" desktop \"\${desktop}\")
  file(WRITE \"\${prefix}/${CMAKE_INSTALL_DATADIR}/applications/org.kevcpp.KevCpp.desktop\" \"\${desktop}\")
  message(STATUS \"Installing: \${prefix}/${CMAKE_INSTALL_DATADIR}/applications/org.kevcpp.KevCpp.desktop\")
  # Let the menu and panels pick up the new launcher and icon right away (best effort).
  execute_process(COMMAND gtk-update-icon-cache -f -t \"\${prefix}/${CMAKE_INSTALL_DATADIR}/icons/hicolor\" OUTPUT_QUIET ERROR_QUIET)
  execute_process(COMMAND update-desktop-database \"\${prefix}/${CMAKE_INSTALL_DATADIR}/applications\" OUTPUT_QUIET ERROR_QUIET)
  # File-manager icon for the installed binary too (see the POST_BUILD step above).
  execute_process(COMMAND gio set \"\${prefix}/${CMAKE_INSTALL_BINDIR}/kev-cpp\" metadata::custom-icon
                          \"file://\${prefix}/${CMAKE_INSTALL_DATADIR}/icons/hicolor/256x256/apps/org.kevcpp.KevCpp.png\"
                  OUTPUT_QUIET ERROR_QUIET)
")
install(DIRECTORY tools/ DESTINATION ${CMAKE_INSTALL_DATADIR}/kev-cpp/tools)
# The Boilerplate menu's snippets (MainWindowBoilerplate.cpp finds them next to the binary).
install(DIRECTORY data/boilerplate/ DESTINATION ${CMAKE_INSTALL_DATADIR}/kev-cpp/boilerplate)

# typescript-language-server + typescript for JS/TS IntelliSense, pinned in tools/package-lock.json.
find_program(NPM_EXECUTABLE npm)
if(NPM_EXECUTABLE)
  add_custom_command(
    OUTPUT ${CMAKE_SOURCE_DIR}/tools/node_modules/.package-lock.json
    COMMAND ${NPM_EXECUTABLE} ci --no-audit --no-fund
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}/tools
    DEPENDS ${CMAKE_SOURCE_DIR}/tools/package.json ${CMAKE_SOURCE_DIR}/tools/package-lock.json
    COMMENT "Installing the TypeScript language server (npm ci)")
  add_custom_target(kevc_node_tools ALL DEPENDS ${CMAKE_SOURCE_DIR}/tools/node_modules/.package-lock.json)
  # Microsoft's js-debug for debugging web projects; pinned and hash-checked (see the script).
  add_custom_command(
    OUTPUT ${CMAKE_SOURCE_DIR}/tools/js-debug/.kev-cpp-1.140.0
    COMMAND ${CMAKE_COMMAND} -P ${CMAKE_SOURCE_DIR}/tools/fetch-js-debug.cmake
    DEPENDS ${CMAKE_SOURCE_DIR}/tools/fetch-js-debug.cmake
    COMMENT "Fetching js-debug (the JavaScript debugger)")
  add_custom_target(kevc_js_debug ALL DEPENDS ${CMAKE_SOURCE_DIR}/tools/js-debug/.kev-cpp-1.140.0)
else()
  message(WARNING "npm not found: TypeScript IntelliSense will be unavailable")
endif()
